    Private Function GetTEAssign(bmID As String) As String '8-11-2020 Additional Function

        Try
            Dim ooConn = New SqlConnection(My.Settings.DBConnection)
            ooConn.Open()
            Dim SQLScript As String = "SELECT * FROM ViewTEData Where bmID ='" & bmID & "'"
            Dim DISI As New SqlCommand(SQLScript, ooConn)
            Dim DIAR = (DISI.ExecuteReader)
            Do While DIAR.Read

                If (DIAR("PMConfirmGLOPReqBy").ToString = DIAR("RepairedByEnd").ToString) Then
                    Return DIAR("PMConfirmGLOPReqBy").ToString()  'PM ID No.
                Else
                    Return DIAR("PMConfirmGLOPReqBy").ToString() & ", " & DIAR("RepairedByEnd").ToString()
                End If
            Loop
            ooConn.Close()

        Catch e As Exception
        End Try

        Return Nothing


    End Function
