    Private Function GetFileVersionInfo(ByVal filename As String) As Version
        Return Version.Parse(FileVersionInfo.GetVersionInfo(filename).FileVersion)
    End Function
    Private Function CheckVerion() As Boolean
        Try
            Dim dt As New DataTable
            Dim strQuery As String = String.Empty
            strQuery = " Select currentversion from  [eMaintenanceSystemDB].[dbo].[MachineKarte_Version] where SystemName = @SystemName"
            Using cmd = New SqlCommand(strQuery, conn)
                cmd.CommandType = CommandType.Text
                cmd.Parameters.AddWithValue("@SystemName", "E-Karte System")
                Dim da = New SqlDataAdapter(cmd)
                da.Fill(dt)
            End Using

            GlobalVersion = dt.Rows(0).Item(0).ToString
            If dt.Rows(0).Item(0).ToString = Me.ProductVersion Then
                CheckVerion = True
            Else
                CheckVerion = False
            End If

            Return CheckVerion

        Catch ex As Exception
            CheckVerion = False
        End Try
    End Function

    Private Sub UpdatePCStatus()
        Try
            Dim PCName As String = Environment.MachineName
            Dim dt As New DataTable
            Dim strQuery As String = String.Empty
            If CheckIfExisting(PCName) = False Then

                strQuery = " Insert into [eMaintenanceSystemDB].[dbo].[MachineKarte_PCStatus] (PCname,IPAdd,LastOpen,Version,UserType) values (@PCname,@IPAdd,@LastOpen,@Version,@UserType)"
                Using cmd = New SqlCommand(strQuery, conn)
                    cmd.CommandType = CommandType.Text
                    cmd.Parameters.AddWithValue("@PCname", PCName)
                    cmd.Parameters.AddWithValue("@IPAdd", GetIP())
                    cmd.Parameters.AddWithValue("@LastOpen", Date.Now.ToString("MM/dd/yyyy hh:mm:ss"))
                    cmd.Parameters.AddWithValue("@Version", "v." & Me.ProductVersion)
                    cmd.Parameters.AddWithValue("@UserType", "PM")
                    Dim da = New SqlDataAdapter(cmd)
                    da.Fill(dt)
                End Using
            Else
                strQuery = " Update [eMaintenanceSystemDB].[dbo].[MachineKarte_PCStatus] set IPADD=@IPAdd,Lastopen=@Lastopen,Version=@Version,UserType=@UserType where PCname=@PCname"
                Using cmd = New SqlCommand(strQuery, conn)

        Catch ex As Exception

    Private Function CheckIfExisting(PCName_Temp As String) As Boolean
        Dim dt As New DataTable
        Dim strQuery As String = String.Empty
        strQuery = " Select Seqno from  [eMaintenanceSystemDB].[dbo].[MachineKarte_PCStatus] where PCName = @PCname"
        Using cmd = New SqlCommand(strQuery, conn)
            cmd.CommandType = CommandType.Text
            cmd.Parameters.AddWithValue("@PCname", PCName_Temp)
            Dim da = New SqlDataAdapter(cmd)
            da.Fill(dt)
        End Using

        If dt.Rows.Count > 0 Then
            CheckIfExisting = True
        Else
            CheckIfExisting = False
        End If

        Return CheckIfExisting
    End Function

    Private Function GetIP() As String
        Dim IPADDTEMP As String = ""

        Dim hostName As String = Dns.GetHostName()
        Dim ipEntry As IPHostEntry = Dns.GetHostByName(hostName)
        Dim ipAddress As String = ""
        Try
            For i = 0 To 6
                ipAddress += ipEntry.AddressList(i).ToString() & " , "
            Next
        Catch ex As Exception
            ipAddress += "|"
            ipAddress += ipEntry.AddressList(0).ToString()
        End Try




        IPADDTEMP = ipAddress

        Return IPADDTEMP
    End Function

    Private Sub cbFloor_SelectedIndexChanged(sender As Object, e As EventArgs) Handles cbFloor.SelectedIndexChanged

        If cbProcess.Text = "-Select-" Then
            MsgBox("Please select a process.", MsgBoxStyle.Information, "E-Karte System")
            Exit Sub
        End If

        If cbFloor.Text = "-Select-" Then
            Exit Sub
        End If

        If cbFloor.Text = "All" Then
            cbMachine.Items.Clear()
            cbMachine.Items.Add("All")
            cbMachine.SelectedItem = "All"
        Else
            Dim dt As New DataTable
            Dim strQuery As String = String.Empty
            If cbUser.Text = "PM" Or cbUser.Text = "CC" Or cbUser.Text = "EEE" Or cbUser.Text = "VIS" Then

                strQuery = "  select distinct(mcno) as 'Mcno' FROM [eMaintenanceSystemDB].[dbo].[MCList] a inner join 
                      [eMaintenanceSystemDB].[dbo].[Process] b on a.ProcessID = b.ProcessID
                      where b.ProcessName = @Process and a.MCLocation  = @Location order by a.mcno asc"
            Else
                strQuery = "  select distinct(testerno) as 'testerNo' FROM [eMaintenanceSystemDB].[dbo].[TEBMMaintenance]
                      where process = @process and location  = @Location and testerno <> '' order by testerno asc"
            End If



            Using cmd = New SqlCommand(strQuery, conn)
                cmd.CommandType = CommandType.Text
                cmd.Parameters.AddWithValue("@Process", cbProcess.Text)
                cmd.Parameters.AddWithValue("@Location", cbFloor.Text)
                Dim da = New SqlDataAdapter(cmd)
                da.Fill(dt)
            End Using
            If dt.Rows.Count <> 0 Then
                cbMachine.Items.Clear()
                cbMachine.Items.Add("All")
                For i = 0 To dt.Rows.Count - 1
                    If cbUser.Text = "PM" Or cbUser.Text = "CC" Or cbUser.Text = "EEE" Or cbUser.Text = "VIS" Then
                        cbMachine.Items.Add(dt.Rows(i).Item("Mcno").ToString)
                    Else
                        cbMachine.Items.Add(dt.Rows(i).Item("testerno").ToString)
                    End If
                Next
                cbMachine.SelectedItem = "All"
            Else
                cbMachine.Items.Clear()
            End If
        End If

    End Sub

    Function AreNextFiveColumnValuesEqual(currentRow As DataRow, dataTable As DataTable, column As String, startIndex As Integer, columnMcno As String) As Boolean
        For i As Integer = startIndex To startIndex + 4


            If Not currentRow(columnMcno).Equals(dataTable.Rows(i)(columnMcno)) Then
                Return False
            End If

            If Not currentRow(column).Equals(dataTable.Rows(i)(column)) Then
                Return False
            End If

        Next
        Return True
    End Function

    Function CheckifSevendays(dt As DataTable, FontColor As String, count As Integer, index As Integer) As DataTable
        Dim counttemp As Integer = count

        If counttemp = 1 Then

            Dim currentRowDate As DataRow = dt.Rows(index)
            Dim NextRowDate As DataRow


            NextRowDate = dt.Rows(index + 1)


            Dim currentValue As Object = currentRowDate("Timeofrequest")
            Dim nextValue As Object = NextRowDate("Timeofrequest")

            Dim date1 As DateTime = DateTime.ParseExact(currentValue, "MM/dd/yyyy HH:mm:ss", CultureInfo.InvariantCulture)
            Dim date2 As DateTime = DateTime.ParseExact(nextValue, "MM/dd/yyyy HH:mm:ss", CultureInfo.InvariantCulture)
            Dim tempdate As TimeSpan = date1 - date2

            If tempdate.Days <= 7 And tempdate.Days >= 0 Then
                dt.Rows(index).Item("FontColor") = FontColor
                dt.Rows(index + 1).Item("FontColor") = FontColor
            End If

        Else
            Dim Plus As Integer = 0
            For x = 0 To count
                Dim currentRowDate As DataRow = dt.Rows(index)
                Dim NextRowDate As DataRow
                Plus += 1
                If x = count Then
                    NextRowDate = dt.Rows(count)
                    Plus -= 1
                Else

                    NextRowDate = dt.Rows(index + (Plus))
                End If

                Dim currentValue As Object = currentRowDate("Timeofrequest")
                Dim nextValue As Object = NextRowDate("Timeofrequest")

                Dim date1 As DateTime = DateTime.ParseExact(currentValue, "MM/dd/yyyy HH:mm:ss", CultureInfo.InvariantCulture)
                Dim date2 As DateTime = DateTime.ParseExact(nextValue, "MM/dd/yyyy HH:mm:ss", CultureInfo.InvariantCulture)
                Dim tempdate As TimeSpan = date1 - date2

                If tempdate.Days <= 7 And tempdate.Days >= 0 Then
                    If x = 0 Then
                        dt.Rows(index).Item("FontColor") = FontColor
                    End If
                    dt.Rows(index + Plus).Item("FontColor") = FontColor
                End If

            Next
        End If
        Return dt

    End Function

    Sub GetDuplicateRowsByColumn(table As DataTable, columnName As String, FirstColumn As String)
        Dim duplicateRows As New List(Of DataRow)()
        Dim NotduplicateRows As New List(Of DataRow)()
        dtNotEquals = table.Clone()
        Dim dt As DataTable = table.Clone()
        Dim uniqueValues As New HashSet(Of Object)()

        Dim rowCount = table.Rows.Count
        For i As Integer = 0 To rowCount - 2
            Dim currentRow As DataRow = table.Rows(i)
            Dim nextRow As DataRow = table.Rows(i + 1)
            Dim PrevRow As DataRow = Nothing
            Dim currentValue As Object = currentRow(columnName)
            Dim nextValue As Object = nextRow(columnName)
            Dim PrevValue As Object = Nothing

            If currentValue.Equals(nextValue) Then
                duplicateRows.Add(currentRow)
            Else
                If i <> 0 Then
                    PrevRow = table.Rows(i - 1)
                    PrevValue = PrevRow(columnName)
                    If currentValue.Equals(PrevValue) Then
                        duplicateRows.Add(currentRow)
                    Else
                        NotduplicateRows.Add(currentRow)
                    End If
                End If
                uniqueValues.Add(currentValue)
            End If

            If i = rowCount - 2 Then
                Dim lastRow As DataRow = table.Rows(rowCount - 1)
                Dim lastValue As Object = lastRow(columnName)
                If currentValue.Equals(lastValue) Then
                    duplicateRows.Add(lastRow)
                    'Else
                    '    NotduplicateRows.Add(currentRow)
                End If
            End If

        Next

        For Each duplicateRow As DataRow In duplicateRows
            dt.ImportRow(duplicateRow)
        Next

        For Each duplicateRow As DataRow In NotduplicateRows
            dtNotEquals.ImportRow(duplicateRow)
        Next

        AreOtherRowsSame(dt, FirstColumn)
    End Sub

    Private Sub RemoveDuplicates()


        For Each ctrl As ListBox In Panel7.Controls
            If TypeOf ctrl Is ListBox Then

                For i As Integer = 0 To ctrl.Items.Count - 2
                    For j As Integer = ctrl.Items.Count - 1 To i + 1 Step -1
                        If ctrl.Items(i).ToString() = ctrl.Items(j).ToString() Then
                            ctrl.Items.RemoveAt(j)
                        End If
                    Next
                Next

            End If
        Next

    End Sub
