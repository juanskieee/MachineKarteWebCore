    Private Sub LoadData(ByVal bmid As String, ByVal mcno As String)

        Dim dt As New DataTable()


        Dim strQuery As String = String.Empty
        If Form1.cbUser.Text = "TE" Then
            strQuery = " SELECT * FROM [dbo].[MachineKarte_ActionCard_TE] WHERE bmid = @bmid AND Machineno = @Machineno"
        Else
            Dim userTemp As String = String.Empty
            Dim TableTemp As String = String.Empty

            If Form1.cbUser.Text = "CC" Then
                TableTemp = "MachineKarte_ActionCard_CC"
            ElseIf Form1.cbUser.Text = "VIS" Then
                TableTemp = "MachineKarte_ActionCard_VIS"
            ElseIf Form1.cbUser.Text = "EEE" Then
                TableTemp = "MachineKarte_ActionCard_EEE"
            Else
                TableTemp = "MachineKarte_ActionCard"
            End If

            strQuery = " SELECT * FROM [dbo].[" & TableTemp & "] WHERE bmid = @bmid AND Machineno = @Machineno"
        End If


        Using cmd As New SqlCommand(strQuery, conn)
            cmd.CommandType = CommandType.Text
            cmd.Parameters.AddWithValue("@bmid", bmid)
            cmd.Parameters.AddWithValue("@Machineno", mcno)
            Using da As New SqlDataAdapter(cmd)
                da.Fill(dt)
            End Using
        End Using

        If dt.Rows.Count > 0 Then
            Dim row As DataRow = dt.Rows(0)
            tbProblemContent.Text = row("ProblemContent").ToString()
            tbInCharge.Text = row("Incharge").ToString()
            tbDate.Text = row("DateCard").ToString()
            tbInvestigation.Text = row("InvestigationDone").ToString()
            tbAction.Text = row("Action").ToString()
            tbComment.Text = row("Comment").ToString()
            tbdate_1.Text = row("Date1").ToString()
            tbIncharge_1.Text = row("InCharge1").ToString()
            tbComment2.Text = row("Comment2").ToString()
            tbdate_2.Text = row("Date2").ToString()
            tbIncharge_2.Text = row("InCharge2").ToString()
            tbComment3.Text = row("Comment3").ToString()
            tbdate_3.Text = row("Date3").ToString()
            tbIncharge_3.Text = row("InCharge3").ToString()
            cbStatus.SelectedItem = row("MCStatus").ToString()
            tbReply1.Text = row("PMReply1").ToString()
            tbReply2.Text = row("PMReply2").ToString()
            tbReply3.Text = row("PMReply3").ToString()
            tbReplyDate1.Text = row("PMDate1").ToString()
            tbReplyDate2.Text = row("PMDate2").ToString()
            tbReplyDate3.Text = row("PMDate3").ToString()
            tbReplyEmp1.Text = row("PMIncharge1").ToString()
            tbReplyEmp2.Text = row("PMIncharge2").ToString()
            tbReplyEmp3.Text = row("PMIncharge3").ToString()
        Else
            MsgBox("No data.", MsgBoxStyle.Information, "Check And Action Card")
            Me.Close()
        End If
    End Sub

        Else
            SaveData()

    End Sub
