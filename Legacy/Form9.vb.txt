    Private Function getPMCategory(ByVal id As Integer) As String
        Try

            Dim oConn As New SqlConnection
            Dim category As String = "0"

            oConn = New SqlConnection(My.Settings.DBConnection)
            oConn.Open()
            Dim Sql As String = "SELECT PMCategory FROM PMCategory where PMCategoryID = " & id
            Dim DC As New SqlCommand(Sql, oConn)
            Dim DR = (DC.ExecuteReader)

            Do While DR.Read
                category = DR("PMCategory").ToString
            Loop
            oConn.Close()
            Return category

        Catch ex As Exception
            Return "-"
        End Try

    End Function
    Private Function getProcCodes(procCodeID As String, tableName As String, procCodeIDCrit As String, retValue As String) As String
        Try

            Dim oConn As New SqlConnection
            Dim pCodes As String = "0"

            oConn = New SqlConnection(My.Settings.DBConnection)
            oConn.Open()
            Dim Sql As String = "SELECT " & retValue & " FROM " & tableName & " Where " & procCodeIDCrit & "=" & procCodeID & ""
            Dim DC As New SqlCommand(Sql, oConn)
            Dim DR = (DC.ExecuteReader)

            Do While DR.Read
                pCodes = DR(retValue).ToString
            Loop
            oConn.Close()
            Return pCodes

        Catch ex As Exception
            Return "0"
        End Try

    End Function
    Private Function getBMWhyInfo(BMWhyID As String, retField As String) As String
        Dim oConn As New SqlConnection

        oConn = New SqlConnection(My.Settings.DBConnection)
        oConn.Open()

        Dim mSQLStr As String = "SELECT " & retField & " FROM BMWhy WHERE BMWhyID=" & BMWhyID

        Dim emDC As New SqlCommand(mSQLStr, oConn)
        Dim emDR As SqlDataReader = (emDC.ExecuteReader)
        Dim retData As String = ""

        Do While emDR.Read

            retData = emDR(retField).ToString

        Loop
        emDR.Close()
        Return retData
    End Function

    Private Sub TabControl1_DrawItem(sender As Object, e As DrawItemEventArgs)
        Dim tabctl As TabControl = DirectCast(sender, TabControl)
        Dim g As Graphics = e.Graphics
        Dim font As Font = tabctl.Font
        Dim brush As New SolidBrush(Color.Black)
        Dim tabTextArea As RectangleF = RectangleF.op_Implicit(tabctl.GetTabRect(e.Index))

        If tabctl.SelectedIndex = e.Index Then
            font = New Font(font, FontStyle.Bold)
            brush = New SolidBrush(Color.White)
            g.FillRectangle(Brushes.LimeGreen, tabTextArea)
        End If
        g.DrawString(tabctl.TabPages(e.Index).Text, font, brush, tabTextArea)
    End Sub
    Private Function getVRNumber(rank As Integer) As String
        Dim Result As String = ""
        Dim Column As String = "VRNumber"
        Dim oConn As New SqlConnection
        oConn = New SqlConnection(My.Settings.DBConnection)
        oConn.Open()

        Dim Sql As String = "SELECT VRNumber1, VRNumber2, VRNumber3 FROM [eMaintenanceSystemDB].[dbo].[VRNumber] WHERE bmID='" & lblID & "'"
        Dim DC As New SqlCommand(Sql, oConn)
        Dim DR = DC.ExecuteReader

        Do While DR.Read
            Result = DR(Column & rank).ToString
        Loop

        oConn.Close()
        Return Result
    End Function
