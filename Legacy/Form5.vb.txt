    Function ConvertDate(ByVal tempStr As String, ByVal dateYear As Integer) As String
        Dim monthMap As New Dictionary(Of String, String) From {
        {"1", "Jan"}, {"2", "Feb"}, {"3", "Mar"}, {"4", "Apr"},
        {"5", "May"}, {"6", "June"}, {"7", "July"}, {"8", "Aug"},
        {"9", "Sept"}, {"10", "Oct"}, {"11", "Nov"}, {"12", "Dec"}
    }

        Dim monthName As String = monthMap(tempStr)
        Return $"{monthName} {dateYear}"
    End Function

    Function AreNextFiveColumnValuesEqual(currentRow As DataRow, dataTable As DataTable, column As String, startIndex As Integer) As Boolean
        For i As Integer = startIndex To startIndex + 4
            If Not (currentRow(column).Equals(dataTable.Rows(i)(column)) And currentRow("Mcno").Equals(dataTable.Rows(i)("Mcno"))) Then
                Return False

            End If
        Next
        Return True
    End Function
    Function AreNextFourColumnValuesEqual(currentRow As DataRow, dataTable As DataTable, column As String, startIndex As Integer) As Boolean
        For i As Integer = startIndex To startIndex + 3
            If Not (currentRow(column).Equals(dataTable.Rows(i)(column)) And currentRow("Mcno").Equals(dataTable.Rows(i)("Mcno"))) Then
                Return False

            End If
        Next
        Return True
    End Function
    Function AreNextThreeColumnValuesEqual(currentRow As DataRow, dataTable As DataTable, column As String, startIndex As Integer) As Boolean
        For i As Integer = startIndex To startIndex + 2

                Return False
            End If

        Return True
    End Function
    Function AreNextTwoColumnValuesEqual(currentRow As DataRow, dataTable As DataTable, column As String, startIndex As Integer) As Boolean

    Function AreNextOneColumnValuesEqual_MCNO(currentRow As DataRow, dataTable As DataTable, column As String, startIndex As Integer) As Boolean

    Private Function GetTablename(Process As String) As String
        GetTablename = ""
        Select Case Process
            Case "DC"
                GetTablename = "DCBMMaintenance"
            Case "DB"
                GetTablename = "DBBMMaintenance"
            Case "WB"
                GetTablename = "WBBMMaintenance"
            Case "MP"
                GetTablename = "MPBMMaintenance"
            Case "PL"
                GetTablename = "PLBMMaintenance"
            Case "FL"
                GetTablename = "FLBMMaintenance"
            Case "FT"
                GetTablename = "FTBMMaintenance"
            Case "TP"
                GetTablename = "TPBMMaintenance"
            Case "EDS"
                GetTablename = "EDSBMMaintenance"
            Case "MEMS"
                GetTablename = "MEMSBMaintenance"
        End Select

        Return GetTablename
    End Function
