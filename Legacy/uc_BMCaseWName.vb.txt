    Function AreNextThreeColumnValuesEqual(currentRow As DataRow, dataTable As DataTable, column As String, startIndex As Integer) As Boolean
        For i As Integer = startIndex To startIndex + 2
            If Not currentRow(column).Equals(dataTable.Rows(i)(column)) Then
                Return False
            End If
        Next
        Return True
    End Function
    Function AreNextTwoColumnValuesEqual(currentRow As DataRow, dataTable As DataTable, column As String, startIndex As Integer) As Boolean
        For i As Integer = startIndex To startIndex + 1
            If Not currentRow(column).Equals(dataTable.Rows(i)(column)) Then
                Return False
            End If
        Next
        Return True
    End Function
    Function AreNextOneColumnValuesEqual(currentRow As DataRow, dataTable As DataTable, column As String, startIndex As Integer) As Boolean
        For i As Integer = startIndex To startIndex
            If Not currentRow(column).Equals(dataTable.Rows(i)(column)) Then
                Return False
            End If
        Next
        Return True
    End Function
    Function RowsAreEqual(row1 As DataRow, row2 As DataRow) As Boolean
        For Each column As DataColumn In row1.Table.Columns
            If Not row1(column).Equals(row2(column)) Then
                Return False
            End If
        Next
        Return True
    End Function
