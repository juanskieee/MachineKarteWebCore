    Private Sub Label_Click(ByVal sender As System.Object, ByVal e As System.EventArgs)
        Dim myButton = DirectCast(sender, Label)
        Try

    Sub AreOtherRowsSame(table As DataTable, firstColumn As String)
        Dim SolveCase As Boolean = False

        Dim ItemCount As Integer = 0
        Dim rowCount = table.Rows.Count
        Dim newDt As New DataTable


        dtNewRowSame = table.Clone
        newDt = table.Clone


        For i As Integer = 0 To rowCount - 2
            Dim currentRow As DataRow = table.Rows(i)
            Dim nextRow As DataRow = table.Rows(i + 1)
            Dim currentValue As Object = currentRow("BMCase")
            Dim nextValue As Object = nextRow("BMCase")

            If i = 0 Then
                newDt.ImportRow(currentRow)
            End If
            If Not currentValue.Equals(nextValue) Then
                newDt.ImportRow(nextRow)
            End If
        Next

        For x As Integer = 0 To rowCount - 1
            Dim XcurrentRow As DataRow = table.Rows(x)
            Dim XcurrentValue As Object = XcurrentRow("BMCase")
            Dim XcurrentTime As Object = XcurrentRow("Timeofrequest")

            For y As Integer = 0 To newDt.Rows.Count - 1
                Dim YcurrentRow As DataRow = newDt.Rows(y)
                Dim YcurrentValue As Object = YcurrentRow("BMCase")
                Dim YcurrentTime As Object = YcurrentRow("Timeofrequest")

                If YcurrentValue.Equals(XcurrentValue) Then
                    Dim DateNow_temp As DateTime = DateTime.ParseExact(firstColumn, "MM/dd/yyyy HH:mm:ss", CultureInfo.InvariantCulture)
                    Dim dateY As DateTime = DateTime.ParseExact(YcurrentTime, "MM/dd/yyyy HH:mm:ss", CultureInfo.InvariantCulture)
                    Dim daysTemp As TimeSpan = DateNow_temp - dateY

                     If daysTemp.Days >= 7 Then
                        If (table.Rows(x).Item("HaveAction").ToString) = "Yes" Then
                            table.Rows(x).Item("Color") = "Green"
                        End If

                        Exit For
                    Else
                        If Not YcurrentTime.Equals(XcurrentTime) Then
                            Dim date1 As DateTime = DateTime.ParseExact(YcurrentTime, "MM/dd/yyyy HH:mm:ss", CultureInfo.InvariantCulture)
                            Dim date2 As DateTime = DateTime.ParseExact(XcurrentTime, "MM/dd/yyyy HH:mm:ss", CultureInfo.InvariantCulture)
                            Dim tempdate As TimeSpan = date1 - date2
                            If tempdate.Days >= 7 Then

                                If (table.Rows(x).Item("HaveAction").ToString) = "Yes" Then
                                    table.Rows(x).Item("Color") = "Green"
                                End If

                            End If
                        End If
                    End If
                End If
            Next
        Next


        dtNewRowSame = table

    End Sub

    Sub GetDuplicateRowsByColumn(table As DataTable, columnName As String, FirstColumn As String)
        Dim duplicateRows As New List(Of DataRow)()
        Dim NotduplicateRows As New List(Of DataRow)()
        dtNotEquals = table.Clone()
        Dim dt As DataTable = table.Clone()
        Dim uniqueValues As New HashSet(Of Object)()

        Dim rowCount = table.Rows.Count
        For i As Integer = 0 To rowCount - 2
            Dim currentRow As DataRow = table.Rows(i)
            Dim nextRow As DataRow = table.Rows(i + 1)
            Dim PrevRow As DataRow = Nothing
            Dim currentValue As Object = currentRow(columnName)
            Dim nextValue As Object = nextRow(columnName)
            Dim PrevValue As Object = Nothing

            If currentValue.Equals(nextValue) Then
                duplicateRows.Add(currentRow)
            Else
                If i <> 0 Then
                    PrevRow = table.Rows(i - 1)
                    PrevValue = PrevRow(columnName)
                    If currentValue.Equals(PrevValue) Then
                        duplicateRows.Add(currentRow)
                    Else
                        NotduplicateRows.Add(currentRow)
                    End If
                End If
                uniqueValues.Add(currentValue)
            End If

            If i = rowCount - 2 Then
                Dim lastRow As DataRow = table.Rows(rowCount - 1)
                Dim lastValue As Object = lastRow(columnName)
                If currentValue.Equals(lastValue) Then
                    duplicateRows.Add(lastRow)
                    'Else
                    '    NotduplicateRows.Add(currentRow)
                End If
            End If

        Next

        For Each duplicateRow As DataRow In duplicateRows
            dt.ImportRow(duplicateRow)
        Next

        'For Each duplicateRow As DataRow In NotduplicateRows
        '    dtNotEquals.ImportRow(duplicateRow)
        'Next

        AreOtherRowsSame(dt, FirstColumn)
    End Sub

    Function AreNextFourColumnValuesEqual(currentRow As DataRow, dataTable As DataTable, column As String, startIndex As Integer, columnMcno As String) As Boolean
        For i As Integer = startIndex To startIndex + 3


            If Not currentRow(columnMcno).Equals(dataTable.Rows(i)(columnMcno)) Then
                Return False
            End If

            If Not currentRow(column).Equals(dataTable.Rows(i)(column)) Then
                Return False
            End If

        Next
        Return True

    Function CheckifSevendays(dt As DataTable, FontColor As String, count As Integer, index As Integer) As DataTable
        Dim counttemp As Integer = count

        If counttemp = 1 Then

            Dim currentRowDate As DataRow = dt.Rows(index)
            Dim NextRowDate As DataRow


            NextRowDate = dt.Rows(index + 1)


            Dim currentValue As Object = currentRowDate("Timeofrequest")
            Dim nextValue As Object = NextRowDate("Timeofrequest")

            Dim date1 As DateTime = DateTime.ParseExact(currentValue, "MM/dd/yyyy HH:mm:ss", CultureInfo.InvariantCulture)
            Dim date2 As DateTime = DateTime.ParseExact(nextValue, "MM/dd/yyyy HH:mm:ss", CultureInfo.InvariantCulture)
            Dim tempdate As TimeSpan = date1 - date2

            If tempdate.Days <= 7 And tempdate.Days >= 0 Then
                dt.Rows(index).Item("FontColor") = FontColor
                dt.Rows(index + 1).Item("FontColor") = FontColor
            End If

        Else
            Dim Plus As Integer = 0
            For x = 0 To count
                Dim currentRowDate As DataRow = dt.Rows(index)
                Dim NextRowDate As DataRow
                Plus += 1
                If x = count Then
                    NextRowDate = dt.Rows(count)
                    Plus -= 1
                Else

                    NextRowDate = dt.Rows(index + (Plus))
                End If

                Dim currentValue As Object = currentRowDate("Timeofrequest")
                Dim nextValue As Object = NextRowDate("Timeofrequest")

                Dim date1 As DateTime = DateTime.ParseExact(currentValue, "MM/dd/yyyy HH:mm:ss", CultureInfo.InvariantCulture)
                Dim date2 As DateTime = DateTime.ParseExact(nextValue, "MM/dd/yyyy HH:mm:ss", CultureInfo.InvariantCulture)
                Dim tempdate As TimeSpan = date1 - date2

                If tempdate.Days <= 7 And tempdate.Days >= 0 Then
                    If x = 0 Then
                        dt.Rows(index).Item("FontColor") = FontColor
                    End If
                    dt.Rows(index + Plus).Item("FontColor") = FontColor
                End If

            Next
        End If
        Return dt

    End Function

        For Each ctrl As ListBox In Panel5.Controls
